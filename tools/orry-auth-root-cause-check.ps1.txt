function New-Utf8NoBomEncoding {
return New-Object System.Text.UTF8Encoding($false)
}

function Write-Utf8NoBomFile {
param(
[Parameter(Mandatory=$true)][string]$Path,
[Parameter(Mandatory=$true)][AllowEmptyString()][string]$Content
)
$parent = Split-Path -Parent $Path
if ([string]::IsNullOrWhiteSpace($parent) -eq $false -and (Test-Path -LiteralPath $parent) -eq $false) {
New-Item -ItemType Directory -Path $parent -Force | Out-Null
}
[System.IO.File]::WriteAllText($Path, $Content, (New-Utf8NoBomEncoding))
}

function Append-Utf8NoBomLine {
param(
[Parameter(Mandatory=$true)][string]$Path,
[Parameter(Mandatory=$true)][AllowEmptyString()][string]$Line
)
$parent = Split-Path -Parent $Path
if ([string]::IsNullOrWhiteSpace($parent) -eq $false -and (Test-Path -LiteralPath $parent) -eq $false) {
New-Item -ItemType Directory -Path $parent -Force | Out-Null
}
$prefix = ''
if (Test-Path -LiteralPath $Path) {
$prefix = [System.IO.File]::ReadAllText($Path, (New-Utf8NoBomEncoding))
if ($prefix.Length -gt 0 -and -not $prefix.EndsWith("rn")) {
$prefix += "rn"
}
}
$prefix += $Line
if (-not $prefix.EndsWith("rn")) {
$prefix += "rn"
}
[System.IO.File]::WriteAllText($Path, $prefix, (New-Utf8NoBomEncoding))
}

function Log {
param([string]$Message)
$ts = Get-Date -Format 'yyyy-MM-dd HH:mm'
$line = "[{0}] {1}" -f $ts, $Message
Write-Host $line
Append-Utf8NoBomLine -Path $script -Line $line
}

function Step-Start {
param([string]$Name)
$script = $Name
Log ("STEP START: {0}" -f $Name)
}

function Step-Ok {
param([string]$Name, [int]$ExitCode = 0)
Log ("STEP OK: {0} | exit={1}" -f $Name, $ExitCode)
}

function Step-Fail {
param([string]$Name, [string]$ErrorMessage, [int]$ExitCode = 1)
Log ("STEP FAIL: {0} | exit={1} | error={2}" -f $Name, $ExitCode, $ErrorMessage)
throw $ErrorMessage
}

function Backup-FileIfExists {
param(
[Parameter(Mandatory=$true)][string]$FilePath,
[Parameter(Mandatory=$true)][string]$BackupDir
)
if (Test-Path -LiteralPath $FilePath) {
$dest = Join-Path $BackupDir ([IO.Path]::GetFileName($FilePath))
Copy-Item -LiteralPath $FilePath -Destination $dest -Force
Log ("Backup created: {0}" -f $dest)
}
}

function Get-SafeFileText {
param([string]$Path)
try {
if (Test-Path -LiteralPath $Path) {
return [System.IO.File]::ReadAllText($Path)
}
} catch {
return ''
}
return ''
}

function Get-ProjectRefFromUrl {
param([string]$Url)
if ([string]::IsNullOrWhiteSpace($Url)) { return $null }
$m = [regex]::Match($Url, 'https?://([a-z0-9]{20}).supabase.co', 'IgnoreCase')
if ($m.Success) { return $m.Groups[1].Value }
return $null
}

function Get-KeyFingerprint {
param([string]$Value)
if ([string]::IsNullOrWhiteSpace($Value)) { return '(empty)' }
$bytes = [System.Text.Encoding]::UTF8.GetBytes($Value)
$sha = [System.Security.Cryptography.SHA256]::Create()
try {
$hash = $sha.ComputeHash($bytes)
} finally {
$sha.Dispose()
}
$hex = -join ($hash | ForEach-Object { $_.ToString('x2') })
return $hex.Substring(0,12)
}

function Parse-EnvFile {
param([string]$Path)
$map = @{}
if (-not (Test-Path -LiteralPath $Path)) { return $map }
$lines = Get-Content -LiteralPath $Path -ErrorAction Stop
foreach ($raw in $lines) {
if ($null -eq $raw) { continue }
$line = $raw.Trim()
if ($line.Length -eq 0) { continue }
if ($line.StartsWith('#')) { continue }
$idx = $line.IndexOf('=')
if ($idx -lt 1) { continue }
$k = $line.Substring(0,$idx).Trim()
$v = $line.Substring($idx+1).Trim()
if (($v.StartsWith('"') -and $v.EndsWith('"')) -or ($v.StartsWith("'") -and $v.EndsWith("'"))) {
if ($v.Length -ge 2) { $v = $v.Substring(1, $v.Length - 2) }
}
$map[$k] = $v
}
return $map
}

function Find-Files {
param(
[string]$Root,
[string[]]$Patterns
)
$all = New-Object System.Collections.Generic.List[string]
foreach ($p in $Patterns) {
Get-ChildItem -LiteralPath $Root -Recurse -File -Filter $p -ErrorAction SilentlyContinue |
ForEach-Object { $all.Add($_.FullName) }
}
return $all | Sort-Object -Unique
}

if (-not (Test-Path -LiteralPath $RepoRoot)) {

}

$ToolsDir = Join-Path $RepoRoot 'tools'
$LogsDir = Join-Path $ToolsDir 'logs'

$SummaryLog = Join-Path $LogsDir 'orry-auth-root-cause-summary.log'
$LastBackup = Join-Path $ToolsDir 'LAST_BACKUP_DIR.txt'

try {

}
}

Step-Ok 'Validate paths and create folders' 0

Step-Fail 'Validate paths and create folders' $_.Exception.Message 1

$envFiles = @(
(Join-Path $RepoRoot '.env'),
(Join-Path $RepoRoot '.env.local'),
(Join-Path $RepoRoot '.env.development'),
(Join-Path $RepoRoot '.env.development.local'),
(Join-Path $RepoRoot '.env.production'),
(Join-Path $RepoRoot '.env.production.local')

Step-Ok 'Backup env files' 0

$envMaps = @()
$allRefs = New-Object System.Collections.Generic.List[object]
$allUrls = New-Object System.Collections.Generic.List[object]

Step-Start 'Parse env files and collect Supabase values'
try {
foreach ($f in $envFiles) {
if (-not (Test-Path -LiteralPath $f)) { continue }
$map = Parse-EnvFile -Path $f
$envMaps += [pscustomobject]@{
Path = $f
Map = $map
}

    foreach ($name in @('NEXT_PUBLIC_SUPABASE_URL','SUPABASE_URL','VITE_SUPABASE_URL')) {
        if ($map.ContainsKey($name)) {
            $url = [string]$map[$name]
            $ref = Get-ProjectRefFromUrl -Url $url
            $allUrls.Add([pscustomobject]@{
                File = $f
                Name = $name
                Url  = $url
                Ref  = $ref
            })
            if ($ref) {
                $allRefs.Add([pscustomobject]@{
                    Source = $f
                    Name   = $name
                    Ref    = $ref
                })
            }
        }
    }

    foreach ($name in @('NEXT_PUBLIC_SUPABASE_ANON_KEY','SUPABASE_ANON_KEY','VITE_SUPABASE_ANON_KEY','SUPABASE_SERVICE_ROLE_KEY')) {
        if ($map.ContainsKey($name)) {
            $fp = Get-KeyFingerprint -Value ([string]$map[$name])
            Log ("ENV {0} | {1} | fingerprint={2}" -f $f, $name, $fp)
        }
    }
}

if ($allUrls.Count -eq 0) {
    Log 'No Supabase URL found in known env files.'
} else {
    foreach ($u in $allUrls) {
        Log ("SUPABASE URL | file={0} | name={1} | ref={2} | url={3}" -f $u.File, $u.Name, ($u.Ref ?? '(none)'), $u.Url)
    }
}

Step-Ok 'Parse env files and collect Supabase values' 0

} catch {
Step-Fail 'Parse env files and collect Supabase values' $_.Exception.Message 1
}

$scanPatterns = @(
'.ts','.tsx','.js','.jsx','.mjs','.cjs','*.json'
)

$likelyDirs = @(
(Join-Path $RepoRoot 'src'),
(Join-Path $RepoRoot 'app'),
(Join-Path $RepoRoot 'lib'),
(Join-Path $RepoRoot 'utils'),
(Join-Path $RepoRoot 'pages')
) | Where-Object { Test-Path -LiteralPath $_ }

(Join-Path $RepoRoot 'middleware.js'),
(Join-Path $RepoRoot 'proxy.ts'),
(Join-Path $RepoRoot 'proxy.js'),
(Join-Path $RepoRoot 'next.config.js'),
(Join-Path $RepoRoot 'next.config.mjs'),
(Join-Path $RepoRoot 'next.config.ts'),
(Join-Path $RepoRoot 'package.json'),
(Join-Path $RepoRoot 'vercel.json')
)

$sourceFiles = New-Object System.Collections.Generic.List[string]
foreach ($d in $likelyDirs) {
foreach ($f in (Find-Files -Root $d -Patterns $scanPatterns)) {
$sourceFiles.Add($f)
}
}
foreach ($sf in $specialFiles) {
if (Test-Path -LiteralPath $sf) { $sourceFiles.Add($sf) }
}
$sourceFiles = $sourceFiles | Sort-Object -Unique

$interesting = New-Object System.Collections.Generic.List[object]

$keyNames = @(
'NEXT_PUBLIC_SUPABASE_URL',
'SUPABASE_URL',
'NEXT_PUBLIC_SUPABASE_ANON_KEY',
'SUPABASE_ANON_KEY',
'SUPABASE_SERVICE_ROLE_KEY',
'createClient',
'@supabase/supabase-js',
'@supabase/ssr',
'createBrowserClient',
'createServerClient',
'signInWithPassword',
'exchangeCodeForSession',
'getSession',
'getUser',
'cookies(',
'middleware',
'proxy',
'test-signin'
)

try {
foreach ($file in $sourceFiles) {
$text = Get-SafeFileText -Path $file
if ([string]::IsNullOrWhiteSpace($text)) { continue }

    $hit = $false
    foreach ($token in $keyNames) {
        if ($text -match [regex]::Escape($token)) {
            $hit = $true
            break
        }

    $matches = $refRegex.Matches($text)
    foreach ($m in $matches) {
        $allRefs.Add([pscustomobject]@{
            Source = $file
            Name   = 'hardcoded_url'
            Ref    = $m.Groups[1].Value
        })
        $hit = $true
    }

        $interesting.Add([pscustomobject]@{
            File = $file
            Size = $text.Length
        })
    }
}

Step-Ok 'Scan source files for auth wiring and hardcoded refs' 0

Step-Fail 'Scan source files for auth wiring and hardcoded refs' $_.Exception.Message 1

foreach ($token in $keyNames) {
if ($line -match [regex]::Escape($token)) {

$end = [Math]::Min($lines.Count - 1, $i + 1)
for ($j = $start; $j -le $end; $j++) {
$snippet = $lines[$j].Trim()
if ($snippet.Length -gt 220) {
$snippet = $snippet.Substring(0,220)
}
Log ("EVIDENCE | file={0} | line={1} | {2}" -f $file, ($j + 1), $snippet)
}
break
}
}
}
}
Step-Ok 'Extract focused auth evidence' 0

Step-Fail 'Extract focused auth evidence' $_.Exception.Message 1

foreach ($line in $summary) {
    Log ("SUMMARY | {0}" -f $line)

Step-Ok 'Build summary' 0

} catch {
Step-Fail 'Build summary' $_.Exception.Message 1

Log ('DONE | log={0}' -f $script)
Log ('DONE | summary={0}' -f $SummaryLog)
exit 0
